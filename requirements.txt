# Runtime dependencies for the Session Pro Backend.
#
# The deployment installs this into a --system-site-packages virtualenv, so pip treats whatever
# Debian already provides as "already satisfied" and installs ONLY what the system lacks. That is
# why flask and pynacl are left unpinned — the Debian-packaged versions are fine and get skipped on
# the server; a plain dev venv just installs current releases.
#
# The DB layer is raw psycopg 3 (+ its connection pool); psycopg/psycopg-pool carry lower bounds for
# the 3.x API the code relies on (conn.transaction(), pool check_connection). The Apple/Google SDKs
# use compatible-release (~=) ranges so fixes flow in without a breaking major.
#
# Dev/test-only extras (pytest, a pip uWSGI for the local example) live in requirements-dev.txt.
flask
pynacl
# Colourised console logging. Debian ships python3-coloredlogs, so the server install skips this.
coloredlogs
# Instants and durations are pendulum's, not the stdlib's — see base.py's module docstring. Bounded to
# 3.x: `Duration`/`Interval` and the exact-vs-calendar arithmetic this depends on are 3.x semantics.
# Debian ships 3.0 (trixie) / 3.2 (sid) as python3-pendulum, so the server install skips this.
pendulum~=3.0
psycopg~=3.1
psycopg-pool~=3.2
app-store-server-library~=1.9
google-cloud-pubsub~=2.31
google-api-python-client~=2.181
